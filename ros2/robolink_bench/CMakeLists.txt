cmake_minimum_required(VERSION 3.8)
project(robolink_bench)
set(CMAKE_CXX_STANDARD 20)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(robolink_msgs REQUIRED)
add_executable(bench_cpp src/bench_cpp.cpp)
target_compile_options(bench_cpp PRIVATE -Wall -Wextra -Werror)
ament_target_dependencies(bench_cpp rclcpp robolink_msgs)
install(TARGETS bench_cpp DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/bench_py.py DESTINATION lib/${PROJECT_NAME})
ament_package()
